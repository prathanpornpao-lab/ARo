        // --- GAME STATE ---
        const gameState = {
            money: 100,
            inventory: { tomato: 0, carrot: 0, cabbage: 0 },
            selectedSeed: 'tomato',
            plots: [], // Array of plot objects
            maxActivePlots: 6,
            houseLevel: 1,
            buffMultiplier: 1,
            buffEndTime: 0,
            robotBroken: false,
            robotBrokenEndTime: 0
        };

        const CROP_CONFIG = {
            tomato:  { cost: 10, sell: 25,  growTime: 5,  color: 0xff3333 },
            carrot:  { cost: 20, sell: 60,  growTime: 8,  color: 0xff8800 },
            cabbage: { cost: 50, sell: 200, growTime: 12, color: 0x88ff88 }
        };

        // --- AUDIO SYSTEM (Tone.js) ---
        let synths = {};
        
        function initAudio() {
            // Main PolySynth for SFX
            synths.pop = new Tone.PolySynth(Tone.Synth, {
                oscillator: { type: "sine" },
                envelope: { attack: 0.01, decay: 0.1, sustain: 0, release: 0.1 }
            }).toDestination();
            synths.pop.volume.value = -10;

            synths.coin = new Tone.Synth({
                oscillator: { type: "triangle" },
                envelope: { attack: 0.05, decay: 0.2, sustain: 0.2, release: 1 }
            }).toDestination();
            synths.coin.volume.value = -5;

            synths.error = new Tone.Synth({
                oscillator: { type: "sawtooth" },
                envelope: { attack: 0.01, decay: 0.2, sustain: 0, release: 0.2 }
            }).toDestination();
            synths.error.volume.value = -15;

            synths.success = new Tone.PolySynth(Tone.Synth, {
                oscillator: { type: "square" },
                envelope: { attack: 0.01, decay: 0.3, sustain: 0.1, release: 1 }
            }).toDestination();
            synths.success.volume.value = -15;
            
            synths.broken = new Tone.NoiseSynth({
                noise: { type: "brown" },
                envelope: { attack: 0.01, decay: 0.5, sustain: 0 }
            }).toDestination();
            synths.broken.volume.value = -20;
        }

        const sfx = {
            playPop: () => synths.pop && synths.pop.triggerAttackRelease(["C5", "E5"], "16n"),
            playPlant: () => synths.pop && synths.pop.triggerAttackRelease(["G4"], "32n"),
            playCoin: () => {
                if(!synths.coin) return;
                synths.coin.triggerAttackRelease("B5", "16n");
                synths.coin.triggerAttackRelease("E6", "8n", "+0.15");
            },
            playError: () => synths.error && synths.error.triggerAttackRelease("C3", "8n"),
            playSuccess: () => synths.success && synths.success.triggerAttackRelease(["C4", "E4", "G4", "C5"], "4n"),
            playWrong: () => synths.error && synths.error.triggerAttackRelease("G2", "4n"),
            playSpark: () => synths.broken && synths.broken.triggerAttackRelease("16n")
        };

        // --- THREE.JS SETUP ---
        const container = document.getElementById('webgl-container');
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x87CEEB); // Sky color
        scene.fog = new THREE.Fog(0x87CEEB, 20, 60);

        const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.set(0, 18, 28);

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); // optimize performance
        renderer.shadowMap.enabled = true;
        renderer.shadowMap.type = THREE.PCFSoftShadowMap;
        container.appendChild(renderer.domElement);

        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.dampingFactor = 0.05;
        controls.minDistance = 10;
        controls.maxDistance = 50;
        controls.maxPolarAngle = Math.PI / 2.1 - 0.1; // Prevent camera from going under ground

        // Lighting
        const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
        scene.add(ambientLight);

        const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
        dirLight.position.set(10, 20, 10);
        dirLight.castShadow = true;
        dirLight.shadow.mapSize.width = 1024;
        dirLight.shadow.mapSize.height = 1024;
        dirLight.shadow.camera.near = 0.5;
        dirLight.shadow.camera.far = 50;
        dirLight.shadow.camera.left = -20;
        dirLight.shadow.camera.right = 20;
        dirLight.shadow.camera.top = 20;
        dirLight.shadow.camera.bottom = -20;
        scene.add(dirLight);

        // Raycaster for interaction
        const raycaster = new THREE.Raycaster();
        const mouse = new THREE.Vector2();

        
        // Materials
        const mats = {
            grass: new THREE.MeshLambertMaterial({ color: 0x7cfc00 }),
            dirt: new THREE.MeshLambertMaterial({ color: 0x5c4033 }),
            darkDirt: new THREE.MeshLambertMaterial({ color: 0x3d2314 }),
            sand: new THREE.MeshLambertMaterial({ color: 0xeedd82 }),
            wood: new THREE.MeshLambertMaterial({ color: 0x8b5a2b }),
            leaf: new THREE.MeshLambertMaterial({ color: 0x228b22 }),
            stone: new THREE.MeshLambertMaterial({ color: 0x888888 }),
            robot: new THREE.MeshLambertMaterial({ color: 0xcccccc }),
            blue: new THREE.MeshLambertMaterial({ color: 0x3b82f6 })
        };

        // 1. Ocean
        const oceanGeo = new THREE.PlaneGeometry(200, 200, 32, 32);
        oceanGeo.rotateX(-Math.PI / 2);
        const oceanMat = new THREE.MeshPhongMaterial({ 
            color: 0x1ca3ec, 
            transparent: true, 
            opacity: 0.8,
            shininess: 80,
            flatShading: true
        });
        const ocean = new THREE.Mesh(oceanGeo, oceanMat);
        ocean.position.y = -1;
        ocean.receiveShadow = true;
        scene.add(ocean);
        
        // Save initial vertices for wave animation
        const oceanInitialPositions = [];
        const posAttribute = oceanGeo.attributes.position;
        for (let i = 0; i < posAttribute.count; i++) {
            oceanInitialPositions.push({
                x: posAttribute.getX(i),
                y: posAttribute.getY(i),
                z: posAttribute.getZ(i)
            });
        }

        // 2. Island Base
        const islandGroup = new THREE.Group();
        scene.add(islandGroup);

        const islandGeo = new THREE.CylinderGeometry(15, 12, 3, 32);
        const island = new THREE.Mesh(islandGeo, mats.sand);
        island.position.y = 0.5;
        island.receiveShadow = true;
        islandGroup.add(island);

        const grassGeo = new THREE.CylinderGeometry(15, 15, 0.5, 32);
        const grass = new THREE.Mesh(grassGeo, mats.grass);
        grass.position.y = 2.25;
        grass.receiveShadow = true;
        islandGroup.add(grass);

        // 3. Trees
        const trees = [];
        function createTree(x, z) {
            const tree = new THREE.Group();
            
            const trunkGeo = new THREE.CylinderGeometry(0.3, 0.4, 2, 8);
            const trunk = new THREE.Mesh(trunkGeo, mats.wood);
            trunk.position.y = 1;
            trunk.castShadow = true;
            tree.add(trunk);

            const leavesGeo = new THREE.ConeGeometry(1.5, 3, 8);
            const leaves = new THREE.Mesh(leavesGeo, mats.leaf);
            leaves.position.y = 3;
            leaves.castShadow = true;
            tree.add(leaves);

            tree.position.set(x, 2.5, z);
            
            // Randomize slight rotation for variety
            tree.rotation.y = Math.random() * Math.PI;
            
            islandGroup.add(tree);
            trees.push({ mesh: tree, offset: Math.random() * 10 });
        }
        // Place trees around the edge
        for(let i=0; i<8; i++) {
            const angle = (i / 8) * Math.PI * 2;
            const radius = 13;
            createTree(Math.cos(angle) * radius, Math.sin(angle) * radius);
        }

        // 4. Well
        const wellGroup = new THREE.Group();
        wellGroup.position.set(-8, 2.5, -6);
        islandGroup.add(wellGroup);
        
        const wellBaseGeo = new THREE.CylinderGeometry(1, 1, 1, 12);
        const wellBase = new THREE.Mesh(wellBaseGeo, mats.stone);
        wellBase.position.y = 0.5;
        wellBase.castShadow = true;
        wellGroup.add(wellBase);
        
        const wellHoleGeo = new THREE.CylinderGeometry(0.8, 0.8, 1.01, 12);
        const wellHole = new THREE.Mesh(wellHoleGeo, new THREE.MeshBasicMaterial({color: 0x111111}));
        wellHole.position.y = 0.5;
        wellGroup.add(wellHole);
        
        const wellPillarGeo = new THREE.BoxGeometry(0.2, 2, 0.2);
        const p1 = new THREE.Mesh(wellPillarGeo, mats.wood);
        p1.position.set(0.9, 1.5, 0); p1.castShadow = true;
        const p2 = new THREE.Mesh(wellPillarGeo, mats.wood);
        p2.position.set(-0.9, 1.5, 0); p2.castShadow = true;
        wellGroup.add(p1, p2);
        
        const wellRoofGeo = new THREE.ConeGeometry(1.8, 1, 4);
        const wellRoof = new THREE.Mesh(wellRoofGeo, mats.wood);
        wellRoof.position.y = 2.8;
        wellRoof.rotation.y = Math.PI / 4;
        wellRoof.castShadow = true;
        wellGroup.add(wellRoof);


        // 5. Farm Plots (3x4 Grid, max 12)
        const plotsGroup = new THREE.Group();
        plotsGroup.position.set(0, 2.5, 4); // Center of farming area
        islandGroup.add(plotsGroup);

        const plotMeshes = [];
        const plotSize = 2.2;
        
        for (let i = 0; i < 12; i++) {
            const col = i % 4;
            const row = Math.floor(i / 4);
            
            const pGroup = new THREE.Group();
            pGroup.position.set((col - 1.5) * plotSize, 0, (row - 1) * plotSize);
            
            // Soil mesh
            const soilGeo = new THREE.BoxGeometry(2, 0.2, 2);
            const soil = new THREE.Mesh(soilGeo, mats.dirt);
            soil.position.y = 0.1;
            soil.receiveShadow = true;
            soil.userData = { type: 'plot', index: i };
            pGroup.add(soil);
            
            plotsGroup.add(pGroup);
            
            // Data structure
            const plotData = {
                index: i,
                active: i < gameState.maxActivePlots, // Start with 6
                state: 'empty', // empty, growing, ready
                cropType: null,
                progress: 0,
                meshGroup: pGroup,
                soilMesh: soil,
                plantMesh: null,
                uiGroup: null // for 3D progress bar
            };
            gameState.plots.push(plotData);
            plotMeshes.push(soil);
            
            // Hide inactive plots
            if (!plotData.active) {
                pGroup.visible = false;
            }
        }

        // 6. House System
        const houseGroup = new THREE.Group();
        houseGroup.position.set(6, 2.5, -5);
        islandGroup.add(houseGroup);
        
        let smokeParticles = [];
        
        function buildHouse(level) {
            // Clear old house
            while(houseGroup.children.length > 0){ 
                houseGroup.remove(houseGroup.children[0]); 
            }
            smokeParticles = []; // Reset smoke

            const wallsMat = new THREE.MeshLambertMaterial({color: 0xfff0dd});
            const roofMat = new THREE.MeshLambertMaterial({color: 0xdd4444});

            if(level >= 1) {
                // Main body
                const bodyGeo = new THREE.BoxGeometry(4, 3, 4);
                const body = new THREE.Mesh(bodyGeo, wallsMat);
                body.position.y = 1.5;
                body.castShadow = true; body.receiveShadow = true;
                houseGroup.add(body);
                
                // Roof
                const roofGeo = new THREE.ConeGeometry(3.5, 2.5, 4);
                const roof = new THREE.Mesh(roofGeo, roofMat);
                roof.position.y = 4.25;
                roof.rotation.y = Math.PI / 4;
                roof.castShadow = true;
                houseGroup.add(roof);

                // Door
                const doorGeo = new THREE.BoxGeometry(1, 1.5, 0.1);
                const door = new THREE.Mesh(doorGeo, mats.wood);
                door.position.set(0, 0.75, 2.05);
                houseGroup.add(door);
                
                // Chimney
                const chimGeo = new THREE.BoxGeometry(0.6, 1.5, 0.6);
                const chim = new THREE.Mesh(chimGeo, mats.stone);
                chim.position.set(1.2, 3.5, -1);
                houseGroup.add(chim);
            }

            if(level >= 2) {
                // Side extension
                const extGeo = new THREE.BoxGeometry(2.5, 2, 3);
                const ext = new THREE.Mesh(extGeo, wallsMat);
                ext.position.set(-3.25, 1, 0);
                ext.castShadow = true;
                houseGroup.add(ext);
                
                const extRoofGeo = new THREE.ConeGeometry(2.5, 1.5, 4);
                const extRoof = new THREE.Mesh(extRoofGeo, roofMat);
                extRoof.position.set(-3.25, 2.75, 0);
                extRoof.rotation.y = Math.PI / 4;
                extRoof.castShadow = true;
                houseGroup.add(extRoof);
            }
            
            if(level >= 3) {
                // Second floor & fence
                const floor2Geo = new THREE.BoxGeometry(3.5, 2, 3.5);
                const floor2 = new THREE.Mesh(floor2Geo, wallsMat);
                floor2.position.set(0, 4, 0);

        // Plant Generators
        function createPlantMesh(type, stage) {
            const group = new THREE.Group();
            let geo, mat, mesh;
            
            // Scale based on stage (growing vs ready)
            const baseScale = stage === 'ready' ? 1 : 0.3;

            if (type === 'tomato') {
                geo = new THREE.SphereGeometry(0.4 * baseScale, 16, 16);
                mat = new THREE.MeshLambertMaterial({ color: 0xff3333 });
                mesh = new THREE.Mesh(geo, mat);
                mesh.position.y = 0.4 * baseScale;
            } else if (type === 'carrot') {
                geo = new THREE.ConeGeometry(0.3 * baseScale, 0.8 * baseScale, 8);
                mat = new THREE.MeshLambertMaterial({ color: 0xff8800 });
                mesh = new THREE.Mesh(geo, mat);
                mesh.rotation.x = Math.PI; // upside down cone
                mesh.position.y = 0.4 * baseScale;
                
                // Add green leaves on top
                const leafG = new THREE.BoxGeometry(0.1, 0.4*baseScale, 0.1);
                const l1 = new THREE.Mesh(leafG, mats.leaf); l1.position.y = 0.8*baseScale;
                group.add(l1);
            } else if (type === 'cabbage') {
                geo = new THREE.DodecahedronGeometry(0.5 * baseScale);
                mat = new THREE.MeshLambertMaterial({ color: 0x88ff88 });
                mesh = new THREE.Mesh(geo, mat);
                mesh.position.y = 0.5 * baseScale;
            }
            
            if(mesh) {
                mesh.castShadow = true;
                group.add(mesh);
            }
            return group;
        }

        // 3D Progress Bar Generator
        function createProgressBar() {
            const group = new THREE.Group();
            
            const bgGeo = new THREE.PlaneGeometry(1, 0.2);
            const bgMat = new THREE.MeshBasicMaterial({ color: 0x000000, side: THREE.DoubleSide });
            const bg = new THREE.Mesh(bgGeo, bgMat);
            group.add(bg);

            const fgGeo = new THREE.PlaneGeometry(0.96, 0.16);
            // shift geometry so scale affects from left to right
            fgGeo.translate(0.48, 0, 0); 
            const fgMat = new THREE.MeshBasicMaterial({ color: 0x00ff00, side: THREE.DoubleSide, depthTest: false });
            const fg = new THREE.Mesh(fgGeo, fgMat);
            fg.position.set(-0.48, 0, 0.01);
            fg.name = 'fill';
            group.add(fg);
            
            group.position.y = 1.2; // Above plant
            return group;
        }

        // Particle System
        const particles = [];
        function createParticles(x, y, z, type) {
            const count = 15;
            for(let i=0; i<count; i++) {
                const geo = new THREE.BoxGeometry(0.1, 0.1, 0.1);
                const color = type === 'water' ? 0x4aa3ff : (type === 'spark' ? 0xffaa00 : 0xffffff);
                const mat = new THREE.MeshBasicMaterial({ color: color });
                const mesh = new THREE.Mesh(geo, mat);
                mesh.position.set(x, y, z);
                
                // Random velocity
                const vx = (Math.random() - 0.5) * 0.2;
                const vy = (Math.random() * 0.2) + 0.1;
                const vz = (Math.random() - 0.5) * 0.2;
                
                scene.add(mesh);
                particles.push({ mesh, vx, vy, vz, life: 1.0, type });
            }
        }

        // Handle Clicks
        function onPointerDown(event) {
            if(document.getElementById('ui-layer').classList.contains('hidden')) return; // Game not started
            
            // Check if clicking on UI
            if(event.target.closest('.glass-btn') || event.target.closest('#quiz-modal')) return;

            mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
            mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            const intersects = raycaster.intersectObjects(plotMeshes);

            if (intersects.length > 0) {
                const object = intersects[0].object;
                const data = object.userData;

                if (data.type === 'plot') {
                    handlePlotClick(data.index, event.clientX, event.clientY);
                } else if (data.type === 'robot') {
                    handleRobotClick();
                }
            }
        }
        window.addEventListener('pointerdown', onPointerDown);

        function handlePlotClick(index, mouseX, mouseY) {
            const plot = gameState.plots[index];
            if (!plot.active) return; // Locked plot

            if (plot.state === 'empty') {
                // Plant
                const crop = CROP_CONFIG[gameState.selectedSeed];
                if (gameState.money >= crop.cost) {
                    gameState.money -= crop.cost;
                    updateUI();
                    
                    plot.state = 'growing';
                    plot.cropType = gameState.selectedSeed;
                    plot.progress = 0;
                    plot.soilMesh.material = mats.darkDirt; // Wet soil
                    
                    // Add small plant mesh
                    plot.plantMesh = createPlantMesh(plot.cropType, 'growing');
                    plot.meshGroup.add(plot.plantMesh);
                    
                    // Add Progress bar
                    plot.uiGroup = createProgressBar();
                    plot.meshGroup.add(plot.uiGroup);
                    
                    sfx.playPlant();
                    showFloatingText(`-${crop.cost}`, 'text-red-500', mouseX, mouseY - 20);
                } else {
                    sfx.playError();
                    showFloatingText("เงินไม่พอ!", 'text-red-600', mouseX, mouseY);
                }
            } else if (plot.state === 'ready') {
                // Harvest
                gameState.inventory[plot.cropType]++;
                updateUI();
                
                plot.state = 'empty';
                plot.soilMesh.material = mats.dirt; // Dry soil
                plot.cropType = null;
                
                plot.meshGroup.remove(plot.plantMesh);
                plot.plantMesh = null;
                
                sfx.playPop();
                
                // Particles
                const wPos = new THREE.Vector3();
                plot.meshGroup.getWorldPosition(wPos);
                createParticles(wPos.x, wPos.y + 1, wPos.z, 'harvest');
            }
        }

        let correctQuizAnswer = 0;

        function handleRobotClick() {
            const now = Date.now();
            if (gameState.robotBroken) {
                if (now < gameState.robotBrokenEndTime) {
                    sfx.playError();
                    return; // Still broken
                } else {
                    gameState.robotBroken = false; // Fix robot
                }
            }
            if (now < gameState.buffEndTime) {
                sfx.playPop(); // Already buffed, just do a pop
                return;
            }

            openQuiz();
        }

        function openQuiz() {
            const num1 = Math.floor(Math.random() * 20) + 1;
            const num2 = Math.floor(Math.random() * 10) + 1;
            const ops = ['+', '-', '*'];
            const op = ops[Math.floor(Math.random() * ops.length)];
            
            let question = `${num1} ${op} ${num2}`;
            if(op === '+') correctQuizAnswer = num1 + num2;
            else if(op === '-') correctQuizAnswer = num1 - num2;
            else if(op === '*') correctQuizAnswer = num1 * num2;

            document.getElementById('quiz-question').innerText = `${question} = ?`;

            // Generate fake answers
            let answers = [correctQuizAnswer];
            while(answers.length < 4) {
                let fake = correctQuizAnswer + Math.floor(Math.random() * 20) - 10;
                if(!answers.includes(fake) && fake >= 0) answers.push(fake);
            }
            answers.sort(() => Math.random() - 0.5); // Shuffle

            const optsDiv = document.getElementById('quiz-options');
            optsDiv.innerHTML = '';
            answers.forEach(ans => {
                const btn = document.createElement('button');
                btn.className = 'glass-btn py-3 rounded-lg font-bold text-xl hover:bg-white/50 text-gray-800';
                btn.innerText = ans;
                btn.onclick = () => checkQuiz(ans);
                optsDiv.appendChild(btn);
            });

            document.getElementById('quiz-modal').classList.remove('hidden');
            sfx.playPop();
        }

        document.getElementById('close-quiz-btn').onclick = () => {

            sfx.playPop();
        };

        function checkQuiz(answer) {
            document.getElementById('quiz-modal').classList.add('hidden');
            if(answer === correctQuizAnswer) {
                // Correct - BUFF
                gameState.buffMultiplier = 1.5;
                gameState.buffEndTime = Date.now() + 10000; // 10 sec
                sfx.playSuccess();
                
                // Water particles over all active plots
                gameState.plots.forEach(p => {
                    if(p.active) {
                        const pos = new THREE.Vector3();
                        p.meshGroup.getWorldPosition(pos);
                        createParticles(pos.x, pos.y + 3, pos.z, 'water');
                    }
                });
            } else {
                // Wrong - BREAK
                gameState.robotBroken = true;
                gameState.robotBrokenEndTime = Date.now() + 10000; // 10 sec
                sfx.playWrong();
                
                const pos = new THREE.Vector3();
                robotGroup.getWorldPosition(pos);
                createParticles(pos.x, pos.y + 2, pos.z, 'spark');
            }
        }


        const clock = new THREE.Clock();
        
        function animate() {
            requestAnimationFrame(animate);
            const delta = clock.getDelta();
            const time = clock.getElapsedTime();

            // 1. Update Ocean Waves
            const posAttribute = oceanGeo.attributes.position;
            for (let i = 0; i < posAttribute.count; i++) {
                const initZ = oceanInitialPositions[i].z; // mapping geometry Z to world Y since rotated
                const initX = oceanInitialPositions[i].x;
                // Simple sine wave formula
                const wave = Math.sin(initX * 0.5 + time) * 0.2 + Math.cos(initZ * 0.5 + time * 0.8) * 0.2;
                posAttribute.setZ(i, oceanInitialPositions[i].z + wave); // Geometry Z is up/down because of rotation
            }
            posAttribute.needsUpdate = true;

            // 2. Animate Trees (Sway)
            trees.forEach(treeData => {
                treeData.mesh.rotation.z = Math.sin(time * 2 + treeData.offset) * 0.05;
                treeData.mesh.rotation.x = Math.cos(time * 1.5 + treeData.offset) * 0.05;
            });

            // 3. House Smoke Particles
            if(Math.random() < 0.1 && gameState.houseLevel >= 1) {
                // Chimney location approx
                const sx = houseGroup.position.x + (gameState.houseLevel >= 3 ? 1.2 : 1.2);
                const sy = houseGroup.position.y + (gameState.houseLevel >= 3 ? 5.5 : 3.5);
                const sz = houseGroup.position.z - 1;
                
                const geo = new THREE.PlaneGeometry(0.5, 0.5);
                const mat = new THREE.MeshBasicMaterial({color: 0xffffff, transparent: true, opacity: 0.5, side: THREE.DoubleSide});
                const smoke = new THREE.Mesh(geo, mat);
                smoke.position.set(sx, sy + 1, sz);
                // Look at camera
                smoke.quaternion.copy(camera.quaternion);
                scene.add(smoke);
                smokeParticles.push({mesh: smoke, life: 1.0});
            }
            for (let i = smokeParticles.length - 1; i >= 0; i--) {
                let p = smokeParticles[i];
                p.mesh.position.y += delta * 2;
                p.mesh.position.x += Math.sin(time + p.life) * delta;
                p.life -= delta * 0.5;
                p.mesh.material.opacity = p.life * 0.5;
                
                // Ensure smoke always faces camera
                p.mesh.quaternion.copy(camera.quaternion);

                if (p.life <= 0) {
                    scene.remove(p.mesh);
                    smokeParticles.splice(i, 1);
                }
            }

            // 4. Update Particles (Harvest, Sparks, Water)
            for (let i = particles.length - 1; i >= 0; i--) {
                let p = particles[i];
                p.mesh.position.x += p.vx;
                p.mesh.position.y += p.vy;
                p.mesh.position.z += p.vz;
                p.vy -= delta * 0.5; // gravity
                p.life -= delta;
                
                if(p.type === 'water') {
                   p.mesh.rotation.x += 0.1;
                   p.mesh.rotation.y += 0.1;
                }

                if (p.life <= 0) {
                    scene.remove(p.mesh);
                    particles.splice(i, 1);
                }
            }

            // 5. Update Plots (Growth)
            gameState.plots.forEach(plot => {
                if (plot.active && plot.state === 'growing') {
                    // Apply buff multiplier
                    const currentMultiplier = (Date.now() < gameState.buffEndTime) ? 1.5 : 1;
                    gameState.buffMultiplier = currentMultiplier; // update state visually
                    
                    const cropConf = CROP_CONFIG[plot.cropType];
                    plot.progress += delta * currentMultiplier;
                    
                    // Update 3D Progress Bar
                    if (plot.uiGroup) {
                        plot.uiGroup.quaternion.copy(camera.quaternion); // Always face camera
                        const fill = plot.uiGroup.getObjectByName('fill');
                        if (fill) {
                            let ratio = plot.progress / cropConf.growTime;
                            if (ratio > 1) ratio = 1;
                            fill.scale.x = ratio;
                        }
                    }

                    // Check if fully grown
                    if (plot.progress >= cropConf.growTime) {
                        plot.state = 'ready';
                        
                        // Remove progress bar
                        plot.meshGroup.remove(plot.uiGroup);
                        plot.uiGroup = null;
                        
                        // Replace mesh with grown version
                        plot.meshGroup.remove(plot.plantMesh);
                        plot.plantMesh = createPlantMesh(plot.cropType, 'ready');
                        plot.meshGroup.add(plot.plantMesh);
                        
                        // Pop animation (scaling up quickly)
                        plot.plantMesh.scale.set(0.1, 0.1, 0.1);
                        plot.plantMesh.userData.targetScale = 1;
                        
                        sfx.playPop(); // Optional: sound when fully grown
                    }
                }
                
                // Animate pop effect if ready
                if(plot.state === 'ready' && plot.plantMesh) {
                    const s = plot.plantMesh.scale.x;
                    if(s < 1) {
                        const newS = s + delta * 5;
                        plot.plantMesh.scale.set(Math.min(newS, 1), Math.min(newS, 1), Math.min(newS, 1));
                    }
                    // Gentle bounce when ready
                    plot.plantMesh.position.y = (plot.cropType === 'tomato' ? 0.4 : (plot.cropType==='cabbage'?0.5:0)) + Math.sin(time*4)*0.05;
                }
            });

            // 6. Update Robot & Label
            const rLabel = document.getElementById('robot-label');
            const rPos = new THREE.Vector3();
            robotGroup.getWorldPosition(rPos);
            rPos.y += 2.5; // Above head
            const screenPos = getScreenPos(rPos);
            
            // Check if object is behind camera
            rPos.project(camera);
            if(rPos.z > 1) {
                rLabel.style.display = 'none';
            } else {
                rLabel.style.display = 'block';
                rLabel.style.left = `${screenPos.x}px`;
                rLabel.style.top = `${screenPos.y}px`;
            }

            const now = Date.now();
            if (gameState.robotBroken) {
                // Shaking
                robotGroup.rotation.z = Math.sin(time * 30) * 0.1;
                robotGroup.position.x = -8 + Math.sin(time * 20) * 0.05;
                
                // Sparks effect
                if(Math.random() < 0.1) {
                    createParticles(robotGroup.position.x, robotGroup.position.y + 1, robotGroup.position.z, 'spark');
                    sfx.playSpark();
                }

                const remaining = Math.ceil((gameState.robotBrokenEndTime - now) / 1000);
                rLabel.innerText = `พัง! (${remaining}s) ⚠️`;
                rLabel.style.color = 'red';
                rLabel.style.border = '2px solid red';
            } else if (now < gameState.buffEndTime) {
                // Spinning (Buff active)
                robotGroup.rotation.y += delta * 5;
                robotGroup.rotation.z = 0;
                robotGroup.position.x = -8;
                
                const remaining = Math.ceil((gameState.buffEndTime - now) / 1000);
                rLabel.innerText = `รดน้ำ 💦 (${remaining}s)`;
                rLabel.style.color = '#3b82f6';
                rLabel.style.border = '2px solid #3b82f6';
            } else {
                // Idle
                robotGroup.rotation.y = Math.sin(time) * 0.2;
                robotGroup.rotation.z = 0;
                robotGroup.position.x = -8;
                
                rLabel.innerText = 'QUIZ 🤖';
                rLabel.style.color = '#333';
                rLabel.style.border = 'none';
            }

            controls.update();
            renderer.render(scene, camera);
        }

        // Handle Window Resize
        window.addEventListener('resize', onWindowResize, false);

            camera.aspect = window.innerWidth / window.innerHeight;

        // Start animation loop
